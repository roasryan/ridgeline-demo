/**
 * Unplanned Studio — website backend for local service sites (template). v3
 * One copy per client, deployed as a Google Apps Script web app (free).
 *
 *   GET  ?action=slots&days=28   → open start times per day ("08:00", "09:30"…) + the owner's settings
 *   GET  ?action=settings        → the owner's availability settings
 *   POST {action:'db_get'|'db_put'} (owner token) · {action:'portal_get'|'portal_act', j, pin} · {action:'crew_get'|'crew_act', code}  → shared job store (v10)
 *   POST {action:'login'|'reset_request'|'reset_set'|'logout'|'whoami'}  → owner email + password sign-in (v9); owner actions send {token}
 *   POST {action:'settings', token, settings}  → owner saves hours / limits from the website's owner screen
 *   POST (lead JSON from the site) → Sheet row, owner alert, booking on the "<Biz> inspections" calendar,
 *        customer confirmation the way THEY picked (text / email / call → owner calls), GoHighLevel if connected.
 *
 * Owner reschedules by dragging the event in Google Calendar (or deletes it to cancel): the customer is told
 * automatically, again by their chosen method. A daily reminder goes out the evening before.
 *
 * Script properties — see SETUP.md
 *   BIZ_NAME, OWNER_NAME, OWNER_EMAIL, OWNER_PHONE, TZ, OWNER_PIN, OWNER_ALERT (email | sms | both)
 *   CHECK_OWNER_CAL (yes/no), CALENDAR_ID (auto), SHEET_ID (auto), SETTINGS (auto, edited from the website)
 *   Texting (optional): SMS_PROVIDER = twilio | textbee
 *     twilio:  TWILIO_SID, TWILIO_TOKEN, TWILIO_FROM (+1…)      textbee: TEXTBEE_KEY, TEXTBEE_DEVICE
 *   GoHighLevel (optional): GHL_TOKEN, GHL_LOCATION_ID, GHL_CALENDAR_ID
 */

const P = PropertiesService.getScriptProperties();
const cfg = (k, d) => P.getProperty(k) || d || '';
const TZ = () => cfg('TZ', 'America/Chicago');
const GHL = 'https://services.leadconnectorhq.com';

const DEFAULT_SETTINGS = {
  hours: { 0: null, 1: ['08:00', '17:00'], 2: ['08:00', '17:00'], 3: ['08:00', '17:00'], 4: ['08:00', '17:00'], 5: ['08:00', '17:00'], 6: ['09:00', '13:00'] },
  len: 60, step: 60, buffer: 0, noticeH: 0, ahead: 28, maxDay: 4, weekCap: 12, blocked: []
};
function settings() { try { return Object.assign({}, DEFAULT_SETTINGS, JSON.parse(cfg('SETTINGS') || '{}')); } catch (x) { return DEFAULT_SETTINGS; } }

/* ------------------------------------------------------------------ GET */
function doGet(e) {
  const p = (e && e.parameter) || {};
  if (p.action === 'settings') return json({ ok: true, settings: settings() });
  if (p.action === 'status') return json({ ok: true, sms: !!smsReady(), email: true });
  if (p.action === 'slots') {
    const S = settings(), days = Math.min(+p.days || S.ahead, 60), key = 'slots_' + days;
    const cache = CacheService.getScriptCache(), hit = cache.get(key);
    if (hit) return json(JSON.parse(hit));
    const out = { ok: true, source: cfg('GHL_TOKEN') ? 'ghl' : 'gcal', settings: S, days: cfg('GHL_TOKEN') ? ghlSlots(days) : calSlots(days) };
    cache.put(key, JSON.stringify(out), 600);
    return json(out);
  }
  return json({ ok: true, service: 'website backend' });
}
function clearSlotCache() { CacheService.getScriptCache().removeAll(['slots_7', 'slots_14', 'slots_28', 'slots_56', 'slots_60']); }

/* ------------------------------------------------------------------ POST */
function doPost(e) {
  let L;
  try { L = JSON.parse(e.postData.contents); } catch (x) { return json({ ok: false, reason: 'bad_json' }); }


  /* ---- owner sign-in (v9, Oct 8): email + password; password set/reset only with a code emailed to OWNER_EMAIL ---- */
  if (L.action === 'login') return json(ownerLogin(L));
  if (L.action === 'reset_request') return json(ownerResetRequest(L));
  if (L.action === 'reset_set') return json(ownerResetSet(L));
  if (L.action === 'logout') { if (L.token) P.deleteProperty('TOK_' + sha(String(L.token))); return json({ ok: true }); }
  if (L.action === 'whoami') return json({ ok: ownerOk(L) });


  /* ---- shared job store (v10, Oct 8): owner app, crew screen and customer portal all read/write the same jobs ---- */
  if (L.action === 'db_get') { if (!ownerOk(L)) { Utilities.sleep(800); return json({ ok: false, reason: 'auth' }); } return json(dbGet()); }
  if (L.action === 'db_put') { if (!ownerOk(L)) { Utilities.sleep(800); return json({ ok: false, reason: 'auth' }); } return json(dbPut(L)); }
  if (L.action === 'portal_get') return json(portalGet(L));
  if (L.action === 'portal_act') return json(portalAct(L));
  if (L.action === 'crew_get') return json(crewGet(L));
  if (L.action === 'crew_act') return json(crewAct(L));

  if (L.action === 'test') {                          // owner checks the real pipes: one email + one text to the owner
    if (!ownerOk(L)) { Utilities.sleep(800); return json({ ok: false, reason: 'auth' }); }
    const out = { ok: true, email: 'skipped', sms: smsReady() ? 'pending' : 'not set up' };
    try { if (cfg('OWNER_EMAIL')) { MailApp.sendEmail(cfg('OWNER_EMAIL'), 'Test: your website emails work', 'This is a test from your website backend. Customer confirmations and lead alerts will look like this.'); out.email = 'sent'; } } catch (x) { out.email = 'error: ' + x.message; }
    if (smsReady()) { try { sendSms(cfg('OWNER_PHONE'), 'Test from your website: customer confirmation texts work.'); out.sms = 'sent'; } catch (x) { out.sms = 'error: ' + String(x.message).slice(0, 160); } }
    out.quota = MailApp.getRemainingDailyQuota();
    return json(out);
  }
  if (L.action === 'settings') {
    if (!ownerOk(L)) { Utilities.sleep(800); return json({ ok: false, reason: 'auth' }); }
    const s = L.settings || {}, clean = {
      hours: {}, len: clampN(s.len, 15, 240, 60), step: clampN(s.step, 15, 240, 60), buffer: clampN(s.buffer, 0, 120, 0),
      noticeH: clampN(s.noticeH, 0, 168, 3), ahead: clampN(s.ahead, 1, 60, 28), maxDay: clampN(s.maxDay, 1, 20, 4), weekCap: clampN(s.weekCap, 1, 60, 12),
      blocked: (s.blocked || []).filter(d => /^\d{4}-\d{2}-\d{2}$/.test(d)).slice(-200)
    };
    for (let d = 0; d < 7; d++) { const h = s.hours && s.hours[d]; clean.hours[d] = h && /^\d\d:\d\d$/.test(h[0]) && /^\d\d:\d\d$/.test(h[1]) && h[1] > h[0] ? [h[0], h[1]] : null; }
    P.setProperty('SETTINGS', JSON.stringify(clean)); clearSlotCache();
    return json({ ok: true });
  }

  if (L.action === 'leads') {
    if (!ownerOk(L)) { Utilities.sleep(800); return json({ ok: false, reason: 'auth' }); }
    return json({ ok: true, leads: ownerLeads() });
  }
  if (L.hp) return json({ ok: true });
  L.phone = String(L.phone || '').replace(/\D/g, '').slice(-10);
  L.email = String(L.email || '').trim();
  const okE = /^\S+@\S+\.\S+$/.test(L.email);
  if (L.phone.length < 10) L.phone = '';
  if (!okE) L.email = '';
  if (!L.name || (!L.phone && !L.email)) return json({ ok: false, reason: 'missing' });
  if (L.best === 'Email' && !L.email) L.best = L.phone ? 'Text' : 'Email';
  if (L.best !== 'Email' && !L.phone) L.best = 'Email';
  L.best = ['Text', 'Call', 'Email'].indexOf(L.best) >= 0 ? L.best : 'Text';

  const cache = CacheService.getScriptCache(), ck = 'rate_' + (L.phone || L.email), n = +(cache.get(ck) || 0);
  if (n >= 5) return json({ ok: false, reason: 'slow_down' });
  cache.put(ck, String(n + 1), 600);

  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    let appt = null;
    if (L.appt && L.appt.date && L.appt.time) {
      const start = at(L.appt.date, L.appt.time);
      if (!start) return json({ ok: false, reason: 'bad_time' });
      const open = cfg('GHL_TOKEN') ? (ghlSlots(45)[L.appt.date] || []) : calSlots(0, L.appt.date)[L.appt.date] || [];
      if (open.indexOf(L.appt.time) < 0) return json({ ok: false, reason: 'taken' });
      appt = { start, end: new Date(start.getTime() + settings().len * 60e3) };
    }
    let ghl = null;
    if (cfg('GHL_TOKEN')) { try { ghl = ghlSave(L, appt); } catch (x) { ghl = { error: String(x) }; } }
    else if (appt) calBook(L, appt);
    if (appt) clearSlotCache();

    sheetRow(L, appt, ghl);
    const notified = notifyCustomer(appt ? 'booked' : L.club ? 'club' : L.lock ? 'lock' : /^Service inquiry/.test(L.type || '') ? 'inquiry' : 'lead', L, appt);
    ownerAlert(L, appt, notified);
    return json({ ok: true, booked: !!appt, notified });
  } finally { lock.releaseLock(); }
}

/* ------------------------------------------------------------------ availability engine (same rules as the website) */
const toMin = t => { const a = String(t).split(':'); return +a[0] * 60 + +a[1]; };
const hhmm = m => ('0' + Math.floor(m / 60)).slice(-2) + ':' + ('0' + (m % 60)).slice(-2);
function bookingCal() {
  const id = cfg('CALENDAR_ID');
  if (id) { const c = CalendarApp.getCalendarById(id); if (c) return c; }
  const c = CalendarApp.createCalendar(cfg('BIZ_NAME', 'Website') + ' inspections', { color: CalendarApp.Color.YELLOW, timeZone: TZ() });
  P.setProperty('CALENDAR_ID', c.getId());
  return c;
}
function busyEvents(from, to) {
  const cal = bookingCal();
  let evs = cal.getEvents(from, to);
  if (cfg('CHECK_OWNER_CAL', 'yes') === 'yes' && cal.getId() !== CalendarApp.getDefaultCalendar().getId())
    evs = evs.concat(CalendarApp.getDefaultCalendar().getEvents(from, to).filter(ev => String(ev.getTransparency && ev.getTransparency()) !== 'TRANSPARENT'));
  return evs;
}
/* open start times per day; pass onlyDate to check a single day */
function calSlots(days, onlyDate) {
  const S = settings(), now = new Date(), out = {};
  const first = onlyDate ? at(onlyDate, '00:00') : at(fmt(now, 'yyyy-MM-dd'), '00:00');
  const last = onlyDate ? new Date(first.getTime() + 864e5) : new Date(first.getTime() + (days + 1) * 864e5);
  const evs = busyEvents(first, last), earliest = now.getTime() + Math.max(S.noticeH * 3600e3, 60e3), cal = bookingCal().getId();
  for (let d = new Date(first); d < last; d = new Date(d.getTime() + 864e5)) {
    const key = fmt(d, 'yyyy-MM-dd'), dow = +fmt(d, 'u') % 7, h = S.hours[dow];
    if (!h || S.blocked.indexOf(key) >= 0) continue;
    const dayEvs = evs.filter(ev => ev.isAllDayEvent() ? fmt(ev.getAllDayStartDate(), 'yyyy-MM-dd') === key : fmt(ev.getStartTime(), 'yyyy-MM-dd') === key || (ev.getStartTime() < at(key, '23:59') && ev.getEndTime() > at(key, '00:00')));
    if (dayEvs.some(ev => ev.isAllDayEvent())) continue;
    const booked = dayEvs.filter(ev => ev.getOriginalCalendarId && ev.getOriginalCalendarId() === cal && /inspection/i.test(ev.getTitle())).length;
    if (booked >= S.maxDay) continue;
    const times = [];
    for (let m = toMin(h[0]); m + S.len <= toMin(h[1]); m += S.step) {
      const s = at(key, hhmm(m)), e = new Date(s.getTime() + S.len * 60e3);
      if (s.getTime() < earliest) continue;
      const bs = new Date(s.getTime() - S.buffer * 60e3), be = new Date(e.getTime() + S.buffer * 60e3);
      if (dayEvs.some(ev => ev.getStartTime() < be && ev.getEndTime() > bs)) continue;
      times.push(hhmm(m));
    }
    if (times.length) out[key] = times;
  }
  return out;
}
function calBook(L, appt) {
  const ev = bookingCal().createEvent(`Inspection: ${L.name}`, appt.start, appt.end, {
    location: L.address || '',
    description: `Phone: ${L.phone ? pretty(L.phone) : 'none'}\nBest way: ${L.best}${L.best === 'Text' && !L.sms ? ' (no auto-text consent)' : ''}\nEmail: ${L.email || ''}\nIssue: ${L.issue || ''}${L.quote ? `\nInstant quote: ${L.quote.name}, ${L.quote.range}` : ''}\n\nDrag this event to reschedule; the customer is told automatically. Delete it to cancel.`
  });
  P.setProperty('bk_' + ev.getId(), JSON.stringify({ n: L.name, p: L.phone, e: L.email || '', b: L.best, s: !!L.sms, a: L.address || '', t: appt.start.toISOString(), r: 0 }));
}

/* ------------------------------------------------------------------ owner moved / cancelled an event → tell the customer */
function onCalChange() {
  const cal = bookingCal(), props = P.getProperties();
  Object.keys(props).filter(k => k.indexOf('bk_') === 0).forEach(k => {
    const b = JSON.parse(props[k]), id = k.slice(3);
    let ev = null; try { ev = cal.getEventById(id); } catch (x) {}
    const L = { name: b.n, phone: b.p, email: b.e, best: b.b, sms: b.s, address: b.a };
    if (!ev) { if (new Date(b.t) > new Date()) { notifyCustomer('cancelled', L, { start: new Date(b.t) }); } P.deleteProperty(k); clearSlotCache(); return; }
    const s = ev.getStartTime();
    if (s.toISOString() !== b.t) {
      b.t = s.toISOString(); b.r = 0; P.setProperty(k, JSON.stringify(b)); clearSlotCache();
      if (s > new Date()) notifyCustomer('moved', L, { start: s, end: ev.getEndTime() });
    }
    if (new Date(b.t).getTime() < Date.now() - 7 * 864e5) P.deleteProperty(k); // tidy old bookings
  });
}
/* every evening: reminder for tomorrow's inspections */
function sendReminders() {
  const tomorrow = fmt(new Date(Date.now() + 864e5), 'yyyy-MM-dd'), props = P.getProperties();
  Object.keys(props).filter(k => k.indexOf('bk_') === 0).forEach(k => {
    const b = JSON.parse(props[k]);
    if (b.r || fmt(new Date(b.t), 'yyyy-MM-dd') !== tomorrow) return;
    notifyCustomer('reminder', { name: b.n, phone: b.p, email: b.e, best: b.b, sms: b.s, address: b.a }, { start: new Date(b.t) });
    b.r = 1; P.setProperty(k, JSON.stringify(b));
  });
}

/* ------------------------------------------------------------------ SMS providers */
function smsReady() {
  const p = cfg('SMS_PROVIDER');
  return (p === 'twilio' && cfg('TWILIO_SID') && cfg('TWILIO_TOKEN') && cfg('TWILIO_FROM')) || (p === 'textbee' && cfg('TEXTBEE_KEY') && cfg('TEXTBEE_DEVICE'));
}
function sendSms(phone, body) {
  const to = '+1' + String(phone).replace(/\D/g, '').slice(-10), p = cfg('SMS_PROVIDER');
  let r;
  if (p === 'twilio') {
    r = UrlFetchApp.fetch(`https://api.twilio.com/2010-04-01/Accounts/${cfg('TWILIO_SID')}/Messages.json`, {
      method: 'post', muteHttpExceptions: true, payload: { To: to, From: cfg('TWILIO_FROM'), Body: body },
      headers: { Authorization: 'Basic ' + Utilities.base64Encode(cfg('TWILIO_SID') + ':' + cfg('TWILIO_TOKEN')) }
    });
  } else if (p === 'textbee') {
    r = UrlFetchApp.fetch(`https://api.textbee.dev/api/v1/gateway/devices/${cfg('TEXTBEE_DEVICE')}/send-sms`, {
      method: 'post', muteHttpExceptions: true, contentType: 'application/json', headers: { 'x-api-key': cfg('TEXTBEE_KEY') }, payload: JSON.stringify({ recipients: [to], message: body })
    });
  } else throw new Error('No SMS provider');
  if (r.getResponseCode() >= 300) throw new Error(p + ' ' + r.getResponseCode() + ' ' + r.getContentText().slice(0, 200));
}

/* ------------------------------------------------------------------ GoHighLevel (optional) */
function ghlFetch(method, path, body, version) {
  const r = UrlFetchApp.fetch(GHL + path, {
    method, muteHttpExceptions: true, contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + cfg('GHL_TOKEN'), Version: version || '2021-07-28', Accept: 'application/json' },
    payload: body ? JSON.stringify(body) : undefined
  });
  const code = r.getResponseCode(), txt = r.getContentText();
  if (code >= 300) throw new Error('GHL ' + method + ' ' + path + ' → ' + code + ' ' + txt.slice(0, 300));
  return txt ? JSON.parse(txt) : {};
}
function ghlSlots(days) {

  const res = ghlFetch('get', `/calendars/${cfg('GHL_CALENDAR_ID')}/free-slots?startDate=${now.getTime()}&endDate=${end.getTime()}&timezone=${encodeURIComponent(TZ())}`, null, '2021-04-15');
  const out = {};
  Object.keys(res).forEach(day => {
    if (!/^\d{4}-\d{2}-\d{2}$/.test(day) || !res[day] || !res[day].slots) return;
    const t = res[day].slots.map(iso => new Date(iso)).filter(d => d.getTime() >= earliest).map(d => fmt(d, 'HH:mm'));
    if (t.length) out[day] = t;
  });
  return out;
}
